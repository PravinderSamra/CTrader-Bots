PART A - PSEUDO-CODE

# OFI is computed from consecutive BEST QUOTE snapshots. You do NOT need full
# order book depth - Level-1 (best bid/ask price and size) is sufficient, which
# is the paper's practical gift.

PROCEDURE order_flow_contribution(prev_quote, curr_quote):
    # e_n, the contribution of the n-th quote update.
    #
    # BID SIDE: think "did buying pressure increase?"
    #   price up            -> new buyers arrived at a higher price -> +new size
    #   price unchanged     -> net change in the queue              -> +delta size
    #   price down          -> bids pulled/consumed                 -> -old size
    #
    # ASK SIDE is the mirror image, and enters with a MINUS sign.

    IF curr.bid_price > prev.bid_price:  e_bid = +curr.bid_size
    ELIF curr.bid_price == prev.bid_price: e_bid = curr.bid_size - prev.bid_size
    ELSE:                                  e_bid = -prev.bid_size

    IF curr.ask_price < prev.ask_price:  e_ask = +curr.ask_size
    ELIF curr.ask_price == prev.ask_price: e_ask = curr.ask_size - prev.ask_size
    ELSE:                                  e_ask = -prev.ask_size

    RETURN e_bid - e_ask

PROCEDURE OFI(interval k):
    # Aggregate contributions over the interval. Equivalently, in the paper's
    # decomposition:
    #     OFI = L_b - C_b - M_s  -  L_s + C_s + M_b
    # where L = limit orders added, C = cancellations, M = market orders.
    #
    # CRUCIAL PROPERTY: a market SELL and a CANCEL of a BID of the same size
    # are treated as EQUIVALENT, because they have the same effect on the bid
    # queue. Trade-based measures are blind to cancellations, which is why
    # trade imbalance achieves R2 of only 32% against OFI's 65%.
    RETURN SUM of order_flow_contribution over all quote updates in interval k

PROCEDURE DEPTH(interval k):
    # Average of the bid and ask queue sizes over the interval.
    RETURN mean( (bid_size + ask_size) / 2 ) over quote updates in k

PROCEDURE fit_price_impact(intervals):
    # Estimate SEPARATELY PER HALF-HOUR. beta varies intraday, and it varies
    # for an identifiable reason: beta is inversely proportional to depth.
    FOR each half-hour subsample i:
        REGRESS  delta_P_k = alpha_i + beta_i * OFI_k + eps_k
    # Expect: alpha indistinguishable from zero, beta strongly significant,
    #         R^2 around 65%, and beta_i ~ 1/D_i across subsamples.

PART B - PYTHON REFERENCE IMPLEMENTATION

"""
Order flow imbalance from a Level-1 quote stream.

Dependencies: numpy, pandas
Input: quotes -> pd.DataFrame with a DatetimeIndex and columns
       ['bid_price', 'bid_size', 'ask_price', 'ask_size'].
"""

import numpy as np
import pandas as pd


def order_flow_contributions(quotes: pd.DataFrame) -> pd.Series:
    """Per-update order flow contribution e_n.

    Fully vectorised. Returns a Series aligned to `quotes` (first row = 0).
    """
    bp, bs = quotes["bid_price"].values, quotes["bid_size"].values
    ap, asz = quotes["ask_price"].values, quotes["ask_size"].values

    pbp, pbs = np.roll(bp, 1), np.roll(bs, 1)
    pap, pas = np.roll(ap, 1), np.roll(asz, 1)

    # Bid side: +new size if price rose, delta if unchanged, -old size if fell.
    e_bid = np.where(bp > pbp, bs, np.where(bp == pbp, bs - pbs, -pbs))
    # Ask side: mirror image (a FALLING ask is increased selling pressure).
    e_ask = np.where(ap < pap, asz, np.where(ap == pap, asz - pas, -pas))

    e = e_bid - e_ask
    e[0] = 0.0                      # no previous quote for the first row
    return pd.Series(e, index=quotes.index, name="e")


def aggregate_ofi(quotes: pd.DataFrame, freq: str = "10s") -> pd.DataFrame:
    """Aggregate into fixed intervals: OFI, mid-price change, and depth.

    freq='10s' reproduces the paper's sampling grid.

    Returns a frame with columns ['OFI', 'dP', 'depth'].
    """
    q = quotes.copy()
    q["e"] = order_flow_contributions(q)
    q["mid"] = (q["bid_price"] + q["ask_price"]) / 2.0
    q["depth"] = (q["bid_size"] + q["ask_size"]) / 2.0

    grp = q.resample(freq)
    out = pd.DataFrame({
        "OFI": grp["e"].sum(),
        "mid_last": grp["mid"].last(),
        "depth": grp["depth"].mean(),
    })
    out["dP"] = out["mid_last"].diff()
    return out[["OFI", "dP", "depth"]].dropna()


def trade_imbalance(trades: pd.DataFrame, freq: str = "10s") -> pd.Series:
    """TI_k = (volume of buyer-initiated trades) - (volume of seller-initiated).

    trades: DatetimeIndex with columns ['size', 'side'] where side is +1 (buy)
    or -1 (sell). The benchmark OFI is compared against - and beats, 65% vs 32%.
    """
    signed = trades["size"] * trades["side"]
    return signed.resample(freq).sum()


def fit_price_impact(df: pd.DataFrame, window: str = "30min") -> pd.DataFrame:
    """Regress dP on OFI separately per half-hour subsample.

    Returns per-window alpha, beta, t(beta), R^2 and mean depth.

    Estimating a SINGLE beta over a whole day is wrong: beta has strong
    intraday seasonality driven by depth, and pooling averages it away.
    """
    rows = []
    for ts, g in df.groupby(pd.Grouper(freq=window)):
        g = g.dropna()
        if len(g) < 10 or g["OFI"].std(ddof=1) == 0:
            continue
        x, y = g["OFI"].values, g["dP"].values
        X = np.column_stack([np.ones(len(x)), x])
        coef, *_ = np.linalg.lstsq(X, y, rcond=None)
        resid = y - X @ coef

        dof = len(x) - 2
        s2 = (resid @ resid) / dof
        xtx_inv = np.linalg.inv(X.T @ X)
        se_beta = np.sqrt(s2 * xtx_inv[1, 1])
        ss_tot = np.sum((y - y.mean()) ** 2)

        rows.append({
            "window": ts,
            "alpha": coef[0],
            "beta": coef[1],
            "t_beta": coef[1] / se_beta if se_beta > 0 else np.nan,
            "r2": 1.0 - (resid @ resid) / ss_tot if ss_tot > 0 else np.nan,
            "mean_depth": g["depth"].mean(),
            "n": len(g),
        })
    return pd.DataFrame(rows)


def check_depth_scaling(fits: pd.DataFrame) -> dict:
    """Test the paper's beta ~ 1/D prediction.

    Regress ln(beta) on ln(depth); the slope should be close to -1.
    """
    f = fits[(fits["beta"] > 0) & (fits["mean_depth"] > 0)].dropna()
    if len(f) < 5:
        return {"error": "not enough valid windows"}
    x, y = np.log(f["mean_depth"].values), np.log(f["beta"].values)
    X = np.column_stack([np.ones(len(x)), x])
    coef, *_ = np.linalg.lstsq(X, y, rcond=None)
    resid = y - X @ coef
    ss_tot = np.sum((y - y.mean()) ** 2)
    return {
        "log_log_slope": float(coef[1]),
        "expected": -1.0,
        "r2": float(1.0 - (resid @ resid) / ss_tot) if ss_tot > 0 else np.nan,
        "n_windows": int(len(f)),
    }


def compare_ofi_vs_trade_imbalance(df: pd.DataFrame, ti: pd.Series) -> dict:
    """Reproduce the paper's regressions (8a), (8b), (8c).

    Expected (paper, 50 stocks, April 2010):
        (8a) OFI alone : average R^2 = 65%
        (8b) TI alone  : average R^2 = 32%
        (8c) both      : t(TI) falls by a factor of ~4; significant in only
                         31% of subsamples, while OFI stays strong.
    """
    d = df.join(ti.rename("TI"), how="inner").dropna()
    if len(d) < 10:
        return {"error": "insufficient overlapping data"}

    def _ols(X, y):
        coef, *_ = np.linalg.lstsq(X, y, rcond=None)
        resid = y - X @ coef
        dof = max(len(y) - X.shape[1], 1)
        s2 = (resid @ resid) / dof
        se = np.sqrt(np.diag(s2 * np.linalg.inv(X.T @ X)))
        ss_tot = np.sum((y - y.mean()) ** 2)
        r2 = 1.0 - (resid @ resid) / ss_tot if ss_tot > 0 else np.nan
        return coef, coef / se, r2

    y = d["dP"].values
    one = np.ones(len(d))
    _, t_o, r2_o = _ols(np.column_stack([one, d["OFI"]]), y)
    _, t_t, r2_t = _ols(np.column_stack([one, d["TI"]]), y)
    _, t_b, r2_b = _ols(np.column_stack([one, d["OFI"], d["TI"]]), y)

    return {
        "r2_ofi_only": float(r2_o),
        "r2_ti_only": float(r2_t),
        "r2_both": float(r2_b),
        "t_ofi_alone": float(t_o[1]),
        "t_ti_alone": float(t_t[1]),
        "t_ofi_joint": float(t_b[1]),
        "t_ti_joint": float(t_b[2]),
        "ti_t_shrinkage": float(abs(t_t[1]) / abs(t_b[2])) if t_b[2] != 0 else np.nan,
    }

# The pipeline was executed in-session on two synthetic Level-1 quote streams.
# Reporting this honestly, because the result is more limited than a
# "reproduction" and the limitation is itself instructive.
#
# WHAT WAS CONFIRMED:
#   - The code runs end to end: contributions -> 10s aggregation -> per-
#     half-hour regressions -> depth scaling -> OFI vs TI comparison.
#   - beta is positive and strongly significant in every window
#     (median t = 4.9 and 12.3 in the two runs; |t| > 2 in 100% of windows).
#   - The intercept alpha comes out near zero, matching the paper's finding
#     that alpha is indistinguishable from zero (AMD: 0% of subsamples reject).
#   - OFI dominates a noise trade-imbalance series in a joint regression:
#     t(OFI) = 21.5 alone and 21.5 jointly, while t(TI) stays insignificant.
#
# WHAT WAS **NOT** REPRODUCED:
#   - The paper's R^2 of 65%. The synthetic runs gave 0.54 and 0.19. This is
#     EXPECTED and is not evidence for or against the paper: a hand-written
#     quote simulator is not real order flow, and the R^2 is entirely a
#     property of the generator. Only real TAQ data can test the 65% claim.
#   - The 32% figure for trade imbalance. The synthetic trades were
#     independent of price by construction, so TI explained ~0% - that is an
#     artefact of the simulator, not a replication of the paper's comparison.
#   - The beta ~ 1/D depth scaling. Testing it needs genuine cross-sectional
#     variation in depth across windows, which the simulator did not provide.
#
# AN INSTRUCTIVE IMPLEMENTATION FINDING:
#   In the first run the quote prices moved on essentially every update. That
#   makes the PRICE-CHANGE branch of the contribution rule fire constantly, so
#   e_n takes the FULL QUEUE SIZE (~500) rather than the size DELTA (~40).
#   OFI was then inflated by roughly two orders of magnitude and the fitted
#   beta shrank by the same factor (1.9e-06 against a generator coefficient of
#   1e-03).
#
#   THE LESSON: beta's SCALE depends on how often the quote price moves in your
#   data. It is not a transferable constant. Always re-estimate beta on the
#   same instrument, venue, and sampling frequency you intend to trade, and
#   never port a beta from one instrument or timescale to another.
#
# BENCHMARK FOR A REAL REPLICATION (paper, 50 US stocks, TAQ, April 2010,
# 10-second grid):
#     OFI alone            R^2 ~= 65%
#     Trade imbalance      R^2 ~= 32%
#     Joint               t(TI) falls ~4x; significant in only 31% of windows
#     AMD example         beta t = 9.96, R^2 = 64%, alpha t = -0.17
# Landing far from these on REAL data points to PITFALL 1 or 2 below.

PART D - USING OFI IN A MARKET-MAKING QUOTER
--------------------------------------------------------------------------------

# This is the defensive application, and it is the one that does NOT require a
# predictive claim.
#
# The market-making entries in this library (Avellaneda-Stoikov,
# Gueant-Lehalle-Fernandez-Tapia, Guilbaud-Pham) all assume order arrivals are
# UNINFORMATIVE. OFI is the measurement that says otherwise. Adding it:
#
#     reservation_price = mid
#                       - q * inventory_skew          [from the MM model]
#                       + theta * beta * OFI_recent   [adverse-selection skew]
#
# where beta is the fitted impact coefficient (scaled by 1/depth) and theta is
# a damping factor in [0, 1].
#
# READ THE SIGN CAREFULLY. Positive OFI means buying pressure and a rising
# mid. A market maker with a resting ASK is about to be filled and then see the
# price continue upward - that is the adverse fill. The response is to shift
# quotes UP (raise the ask away, raise the bid) so the fill happens at a fairer
# price or not at all.
#
# WIDEN WHEN DEPTH IS THIN. Since beta ~ 1/D, the same OFI moves the price
# further in a thin book. Quote width should respond to DEPTH, not just to
# volatility - a point the pure inventory models miss entirely.
#
# ---- PITFALLS ----
#
# 1. CONTEMPORANEOUS, NOT PREDICTIVE. The 65% R^2 regresses OFI against the
#    price change over THE SAME interval. It is a decomposition, not a
#    forecast. Using it as a standalone directional signal requires separate
#    lead-lag evidence that this paper does not provide. The defensive
#    quote-skew application above is valid precisely because it does not need
#    a forecast.
#
# 2. GET THE PRICE-CHANGE CASES RIGHT. The three-way branch (price up /
#    unchanged / down) is the whole definition. Using a naive
#    (bid_size - ask_size) queue imbalance instead is a DIFFERENT and much
#    weaker variable, and is the most common substitution error.
#
# 3. ESTIMATE beta PER HALF-HOUR. beta has strong intraday seasonality driven
#    by depth. A single daily beta averages away the effect the paper is
#    documenting.
#
# 4. LEVEL-1 IS ENOUGH, BUT ONLY FOR BEST-QUOTE FLOW. OFI as defined sees only
#    events at the best bid and ask. Hidden liquidity and deep-book activity
#    are invisible.
#
# 5. TICK SIZE MATTERS. Where the tick is large relative to volatility, the mid
#    is pinned and quantised and there is little variation for OFI to explain.
#
# 6. THE SQUARE-ROOT LAW MAY BE AN ARTEFACT. The paper derives square-root
#    impact from this linear model plus aggregation and argues it "is not
#    robust and is a statistical artifact due to the aggregation of data." If
#    your execution cost model is calibrated to square-root-of-volume, that is
#    worth revisiting.
